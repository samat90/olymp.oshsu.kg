def sol(s):

    T=[[0]*n for _ in range(n)]; F=[[0]*n for _ in range(n)]
    for i in range(n):
        T[i][i]=int(v[i]=='T'); F[i][i]=int(v[i]=='F')

def brute(s):
    e=s.split()[0]; v=e[0::2]; op=e[1::2]
    def go(i,j):
        if i==j: return [v[i]=='T']
        res=[]
        for m in range(i,j):
            for x in go(i,m):
                for y in go(m+1,j):
                    o=op[m]
                    res.append(x and y if o=='&' else (x or y if o=='|' else x!=y))
        return res
    return sum(go(0,len(v)-1))%M
