=== 71 | Прокладка линий связи | Графы | 2
--- statement
Нужно соединить $n$ сёл линиями связи так, чтобы между любыми двумя сёлами существовал маршрут. Даны $m$ возможных линий с ценой. Найдите наименьшую суммарную цену, или $-1$, если соединить все сёла нельзя.

**Ввод.** В первой строке $n$ и $m$ ($1 \le n \le 10^5$, $0 \le m \le 2\cdot 10^5$). Далее $m$ строк $a\ b\ w$ ($1 \le w \le 10^6$).

**Вывод.** Одно число.
--- sample
4 5
1 2 3
2 3 1
3 4 4
1 4 2
1 3 5
--- code
def gen(r, k):
    n = sz(r, k, 100000, lo=1); W = r.choice([4, 1000, 10**6])
    es = [(a, b, r.randint(1, W)) for a, b in graph(r, n, r.randint(n, 2 * n), conn=k % 4 != 0, k=k)]
    return '%d %d\n' % (n, len(es)) + el(es)

def sol(s):
    v = ints(s); n, m = v[0], v[1]; es = sorted((v[4 + 3 * i], v[2 + 3 * i], v[3 + 3 * i]) for i in range(m)); d = DSU(n); tot = 0; c = 0
    for w, a, b in es:
        if d.u(a, b):
            tot += w; c += 1
    return str(tot if c == n - 1 else -1)

def brute(s):
    v = ints(s); n, m = v[0], v[1]; es = [(v[2 + 3 * i], v[3 + 3 * i], v[4 + 3 * i]) for i in range(m)]; best = INF
    for comb in itertools.combinations(es, n - 1):
        d = DSU(n)
        if all(d.u(a, b) for a, b, w in comb):
            best = min(best, sum(w for a, b, w in comb))
    return str(-1 if best == INF else best)

    for w, a, b in es:

Дан связный граф из $n$ вершин и $m$ рёбер с весами. Остов — набор из $n-1$ рёбер, связывающий все вершины. Рассмотрим все остовы наименьшего возможного веса. **Ввод.** В первой строке $n$ и $m$ ($2 \le n \le 100$, $n-1 \le m \le 200$). Далее $m$ строк $a\ b\ w$ ($1 \le w \le 1000$). Граф связный, без кратных рёбер.

--- code
def gen(r, k):
    n = sz(r, k, 100, lo=2, tiny=6); W = r.choice([2, 3, 10, 1000]) if k else 3
    m = r.randint(n - 1, min(200, int(1.8 * n)))
    es = [(a, b, r.randint(1, W)) for a, b in graph(r, n, m, conn=True, k=k)]
    if k == 0:
        es = es[:9] if len(es) > 9 else es
        if not is_connected(n, es):
            es = [(a, b, r.randint(1, W)) for a, b in graph(r, n, n - 1, conn=True, k=k)]
    return '%d %d\n' % (n, len(es)) + el(es)

def mst_w(n, es, skip=-1, force=-1):
    d = DSU(n); tot = 0; c = 0
    if force >= 0:
        a, b, w = es[force]; d.u(a, b); tot += w; c += 1
    for i in sorted(range(len(es)), key=lambda i: es[i][2]):
        if i == skip or i == force:
            continue
        a, b, w = es[i]
        if d.u(a, b):
            tot += w; c += 1
    return tot if c == n - 1 else INF

def sol(s):
    v = ints(s); n, m = v[0], v[1]; es = [(v[2 + 3 * i], v[3 + 3 * i], v[4 + 3 * i]) for i in range(m)]
    W = mst_w(n, es); a = b = 0
    for i in range(m):
        if mst_w(n, es, skip=i) > W:
            a += 1
        elif mst_w(n, es, force=i) == W:
            b += 1
    return '%d %d' % (a, b)

def brute(s):
    v = ints(s); n, m = v[0], v[1]; es = [(v[2 + 3 * i], v[3 + 3 * i], v[4 + 3 * i]) for i in range(m)]
    trees = []
    for comb in itertools.combinations(range(m), n - 1):
        d = DSU(n)
        if all(d.u(es[i][0], es[i][1]) for i in comb):
            trees.append((sum(es[i][2] for i in comb), set(comb)))
    W = min(t[0] for t in trees); good = [t[1] for t in trees if t[0] == W]
    a = sum(1 for i in range(m) if all(i in t for t in good)); b = sum(1 for i in range(m) if any(i in t for t in good)) - a
    return '%d %d' % (a, b)

=== 74 | Запасной остов | Графы | 2
--- statement
Дан связный граф с весами рёбер. Остов — набор из $n-1$ рёбер, связывающий все вершины, его вес — сумма весов рёбер. Выпишем веса всех различных остовов (разные наборы рёбер) по неубыванию. Выведите второе число в этом списке.

**Ввод.** В первой строке $n$ и $m$ ($2 \le n \le 200$, $n-1 \le m \le 1000$). Далее $m$ строк $a\ b\ w$ ($1 \le w \le 1000$). Граф связный, без кратных рёбер.

**Вывод.** Второе число списка или $-1$, если остов единственный (граф — дерево).
--- sample
4 5
1 2 1
2 3 2
3 4 3
1 4 4
1 3 5
--- code
def gen(r, k):
    n = sz(r, k, 200, lo=2, tiny=6); W = r.choice([2, 5, 1000]) if k else 4
    m = r.randint(n - 1, min(1000, 3 * n)) if k else r.randint(n - 1, 8)
    es = [(a, b, r.randint(1, W)) for a, b in graph(r, n, m, conn=True, k=k)]
    return '%d %d\n' % (n, len(es)) + el(es)

def sol(s):
    v = ints(s); n, m = v[0], v[1]; es = [(v[2 + 3 * i], v[3 + 3 * i], v[4 + 3 * i]) for i in range(m)]
    order = sorted(range(m), key=lambda i: es[i][2]); d = DSU(n); inT = set(); tot = 0
    for i in order:
        if d.u(es[i][0], es[i][1]):
            inT.add(i); tot += es[i][2]
    if m == n - 1:
        return '-1'
    g = [[] for _ in range(n + 1)]
    for i in inT:
        a, b, w = es[i]; g[a].append((b, w)); g[b].append((a, w))
    best = INF
    mx = [[0] * (n + 1) for _ in range(n + 1)]
    for s0 in range(1, n + 1):
        st = [(s0, 0, 0)]
        while st:
            x, p, cur = st.pop(); mx[s0][x] = cur
            for y, w in g[x]:
                if y != p:
                    st.append((y, x, max(cur, w)))
    for i in range(m):
        if i not in inT:
            a, b, w = es[i]; best = min(best, tot + w - mx[a][b])
    return str(best)

def brute(s):
    v = ints(s); n, m = v[0], v[1]; es = [(v[2 + 3 * i], v[3 + 3 * i], v[4 + 3 * i]) for i in range(m)]; ws = []
    for comb in itertools.combinations(range(m), n - 1):
        d = DSU(n)
        if all(d.u(es[i][0], es[i][1]) for i in comb):
            ws.append(sum(es[i][2] for i in comb))
    ws.sort()
    return str(ws[1]) if len(ws) > 1 else '-1'
