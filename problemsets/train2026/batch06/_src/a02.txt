**Ввод.** В первой строке m и X (1 ≤ m ≤ 20, 1 ≤ X ≤ 5000), во второй m различных номиналов (1 ≤ c_i ≤ 5000).

**Вывод.** Число способов по модулю 10^9 + 7.

def sol(s):
    t=list(map(int,s.split())); m,X=t[0],t[1]; c=t[2:2+m]

    for v in c:

def brute(s):
    t=list(map(int,s.split())); m,X=t[0],t[1]; c=t[2:2+m]

def sol(s):
    n=int(s.split()[0]); dp=[0]+[10**9]*n
    for x in range(1,n+1):
        j=1
        while j*j<=x:
            dp[x]=min(dp[x],dp[x-j*j]+1); j+=1
    return dp[n]
#B
def brute(s):
    n=int(s.split()[0])
    sq=[j*j for j in range(1,int(n**.5)+1)]
    def can(rem,d,mn):
        if rem==0: return True
        if d==0: return False
        return any(q>=mn and can(rem-q,d-1,q) for q in sq if q<=rem)
    d=1
    while not can(n,d,0): d+=1
    return d
