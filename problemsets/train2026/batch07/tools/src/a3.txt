    return '%d %d %d\n' % (len(a), r.randint(0, 20 if k == 0 else 1000), t) + ' '.join(map(str, a))

def sol(s):
    v = ints(s); n, st, t = v[:3]; a = v[3:]
    if st == t:
        return '0'
    d = {st: 0}; q = deque([st])
    while q:
        x = q.popleft()
        for z in a:
            for y in (x + z, x - z, x ^ z):
                if y == t:
                    return str(d[x] + 1)
                if 0 <= y <= 1000 and y not in d:
                    d[y] = d[x] + 1; q.append(y)
    return '-1'

def brute(s):
    v = ints(s); n, st, t = v[:3]; a = v[3:]
    if st == t:
        return '0'
    d = [INF] * 1001; d[st] = 0
    for _ in range(1002):
        for x in range(1001):
            if d[x] < INF:
                for z in a:
                    for y in (x + z, x - z, x ^ z):
                        if 0 <= y <= 1000:
                            d[y] = min(d[y], d[x] + 1)
    best = min([d[x] + 1 for x in range(1001) for z in a if d[x] < INF and t in (x + z, x - z, x ^ z)] or [INF])
    return str(-1 if best == INF else best)
