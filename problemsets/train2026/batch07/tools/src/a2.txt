def gen(r, k):

    g = grid(r, h, w, r.choice([.1, .18, .25, .3]))

def sol(s):

def gen(r, k):
    h, w = hw(r, k, 300000)
    pe = r.choice([0, .05, .2]); pr = r.choice([.002, .01, .05]) if k else .15
    g = []
    for i in range(h):
        row = ''
        for j in range(w):
            x = r.random()
            row += '0' if x < pe else '2' if x < pe + pr else '1'
        g.append(row)
    return '%d %d\n' % (h, w) + '\n'.join(g)

def sol(s):
    h, w, g = rg(s)
    d = {}; q = deque(); fresh = 0
    for i in range(h):
        for j in range(w):
            if g[i][j] == '2':
                d[(i, j)] = 0; q.append((i, j))
            elif g[i][j] == '1':
                fresh += 1
    last = 0
    while q:
        x, y = q.popleft()
        for a, b in D4:
            c = (x + a, y + b)
            if 0 <= c[0] < h and 0 <= c[1] < w and g[c[0]][c[1]] == '1' and c not in d:
                d[c] = d[(x, y)] + 1; last = d[c]; fresh -= 1; q.append(c)
    return str(last if fresh == 0 else -1)

def brute(s):
    h, w, g = rg(s)
    g = [list(x) for x in g]; t = 0
    while True:
        if not any('1' in row for row in g):
            return str(t)
        new = [row[:] for row in g]; ch = False
        for i in range(h):
            for j in range(w):
                if g[i][j] == '1' and any(0 <= i + a < h and 0 <= j + b < w and g[i + a][j + b] == '2' for a, b in D4):
                    new[i][j] = '2'; ch = True
        if not ch:
            return '-1'
        g = new; t += 1

--- statement
На карте $h \times w$ клетка `1` — колодец, `0` — обычная клетка. Для каждой клетки найдите расстояние до ближайшего колодца, если ходить по сторонам клеток (расстояние между $(i,j)$ и $(a,b)$ равно $|i-a|+|b-j|$). Колодец хотя бы один.

**Ввод.** В первой строке $h$ и $w$ ($1 \le h\cdot w \le 2\cdot 10^5$). Далее $h$ строк по $w$ символов `0`/`1`.

--- code
def gen(r, k):
    h, w = hw(r, k, 200000)
    p = r.choice([.001, .01, .1, .3])
    g = [''.join('1' if r.random() < p else '0' for _ in range(w)) for _ in range(h)]
    setc(g, r.randrange(h), r.randrange(w), '1')
    return '%d %d\n' % (h, w) + '\n'.join(g)

def sol(s):
    h, w, g = rg(s)
    d = [[-1] * w for _ in range(h)]; q = deque()
    for i in range(h):
        for j in range(w):
            if g[i][j] == '1':
                d[i][j] = 0; q.append((i, j))
    while q:
        x, y = q.popleft()
        for a, b in D4:
            c, e = x + a, y + b
            if 0 <= c < h and 0 <= e < w and d[c][e] < 0:
                d[c][e] = d[x][y] + 1; q.append((c, e))
    return '\n'.join(' '.join(map(str, row)) for row in d)

def brute(s):
    h, w, g = rg(s)
    ones = [(i, j) for i in range(h) for j in range(w) if g[i][j] == '1']
    return '\n'.join(' '.join(str(min(abs(i - a) + abs(j - b) for a, b in ones)) for j in range(w)) for i in range(h))

Карта $h \times w$: `#` — скала, `.` — свободная клетка, `G` — ворота. Для каждой свободной клетки найдите наименьшее число ходов (по сторонам, только по клеткам без скал) до ближайших ворот.

**Вывод.** $h$ строк по $w$ значений через пробел: для скалы `#`, для ворот `0`, для свободной клетки расстояние или $-1$, если до ворот не дойти.
--- sample

G.#.
..#.
#..G
--- code
def gen(r, k):
    h, w = hw(r, k, 200000)
    g = grid(r, h, w, r.choice([.15, .3, .4]))
    for _ in range(r.randint(1, max(1, h * w // 500 + 1))):
        setc(g, r.randrange(h), r.randrange(w), 'G')
    return '%d %d\n' % (h, w) + '\n'.join(g)

def sol(s):
    h, w, g = rg(s)
    d = [[-1] * w for _ in range(h)]; q = deque()
    for i in range(h):
        for j in range(w):
            if g[i][j] == 'G':
                d[i][j] = 0; q.append((i, j))
    while q:
        x, y = q.popleft()
        for a, b in D4:
            c, e = x + a, y + b
            if 0 <= c < h and 0 <= e < w and g[c][e] == '.' and d[c][e] < 0:
                d[c][e] = d[x][y] + 1; q.append((c, e))
    return '\n'.join(' '.join('#' if g[i][j] == '#' else str(d[i][j]) for j in range(w)) for i in range(h))

def brute(s):
    h, w, g = rg(s)
    best = {}
    for i in range(h):
        for j in range(w):
            if g[i][j] == 'G':
                d = {(i, j): 0}; ch = True
                while ch:
                    ch = False
                    for (x, y), v in list(d.items()):
                        for a, b in D4:
                            c = (x + a, y + b)
                            if 0 <= c[0] < h and 0 <= c[1] < w and g[c[0]][c[1]] != '#' and d.get(c, 10**9) > v + 1:
                                d[c] = v + 1; ch = True
                for c, v in d.items():
                    best[c] = min(best.get(c, 10**9), v)
    return '\n'.join(' '.join('#' if g[i][j] == '#' else str(best.get((i, j), -1)) for j in range(w)) for i in range(h))

=== 19 | Куда течёт вода | Графы | 1

**Ввод.** В первой строке $h$ и $w$ ($1 \le h\cdot w \le 10^5$). Далее $h$ строк по $w$ целых чисел от $0$ до $10^9$.

def gen(r, k):
    h, w = hw(r, k, 100000)
    hi = r.choice([3, 10, 1000])
    return '%d %d\n' % (h, w) + '\n'.join(' '.join(str(r.randint(0, hi)) for _ in range(w)) for _ in range(h))

def sol(s):
    L = s.split('\n'); h, w = map(int, L[0].split()); g = [list(map(int, L[1 + i].split())) for i in range(h)]
    def go(starts):
        seen = set(starts); st = list(starts)
        while st:
            i, j = st.pop()
            for a, b in D4:
                c = (i + a, j + b)
                if 0 <= c[0] < h and 0 <= c[1] < w and c not in seen and g[c[0]][c[1]] >= g[i][j]:
                    seen.add(c); st.append(c)
        return seen
    A = [(0, j) for j in range(w)] + [(i, 0) for i in range(h)]
    B = [(h - 1, j) for j in range(w)] + [(i, w - 1) for i in range(h)]
    return str(len(go(A) & go(B)))

def brute(s):
    L = s.split('\n'); h, w = map(int, L[0].split()); g = [list(map(int, L[1 + i].split())) for i in range(h)]
    res = 0
    for i in range(h):
        for j in range(w):
            seen = {(i, j)}; ch = True
            while ch:
                ch = False
                for (x, y) in list(seen):
                    for a, b in D4:
                        c = (x + a, y + b)
                        if 0 <= c[0] < h and 0 <= c[1] < w and c not in seen and g[c[0]][c[1]] <= g[x][y]:
                            seen.add(c); ch = True
            ok1 = any(x == 0 or y == 0 for x, y in seen)
            ok2 = any(x == h - 1 or y == w - 1 for x, y in seen)
            res += ok1 and ok2
    return str(res)
