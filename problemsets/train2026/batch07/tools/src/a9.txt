=== 81 | Самый длинный круг | Графы | 1
--- statement
В стране $n$ городов; из города $i$ ведёт не более одной односторонней дороги — в город $a_i$ ($a_i = 0$, если дороги нет; $a_i \ne i$). Круг — замкнутый маршрут, возвращающийся в исходный город, без повторов городов. Найдите длину самого длинного круга.

**Ввод.** В первой строке $n$ ($2 \le n \le 10^5$). Во второй — $n$ чисел $a_i$ ($0 \le a_i \le n$).

**Вывод.** Длина круга или $-1$, если кругов нет.
--- sample
6
2 3 1 3 0 4
--- code
def gen(r, k):
    n = sz(r, k, 100000, lo=2)
    a = []
    p0 = r.choice([0, .1, .4])
    for i in range(1, n + 1):
        if r.random() < p0:
            a.append(0)
        else:
            x = r.randint(1, n)
            while x == i:
                x = r.randint(1, n)
            a.append(x)
    return '%d\n' % n + ' '.join(map(str, a))

def sol(s):
    v = ints(s); n = v[0]; a = [0] + v[1:]; st = [0] * (n + 1); pos = [0] * (n + 1); best = -1
    for s0 in range(1, n + 1):
        if st[s0]:
            continue
        x = s0; t = 0
        while x and not st[x]:
            st[x] = s0; pos[x] = t; t += 1; x = a[x]
        if x and st[x] == s0:
            best = max(best, t - pos[x])
    return str(best)

def brute(s):
    v = ints(s); n = v[0]; a = [0] + v[1:]; best = -1
    for s0 in range(1, n + 1):
        x = s0
        for _ in range(n + 1):
            if x == 0:
                break
            x = a[x]
        if x == 0:
            continue
        y = a[x]; L = 1
        while y != x:
            y = a[y]; L += 1
        best = max(best, L)
    return str(best)

=== 82 | Сколько городов посетим | Графы | 1

**Вывод.** $n$ чисел через пробел.

--- code
def gen(r, k):
    n = sz(r, k, 100000, lo=1)
    if k % 3 == 0:
        a = [r.randint(1, n) for _ in range(n)]
    else:
        a = [max(1, i - r.randint(0, 2)) if r.random() < .98 else r.randint(1, n) for i in range(1, n + 1)]
    return '%d\n' % n + ' '.join(map(str, a))

def sol(s):
    v = ints(s); n = v[0]; a = [0] + v[1:]; res = [0] * (n + 1); st = [0] * (n + 1)
    for s0 in range(1, n + 1):
        if res[s0]:
            continue
        path = []; x = s0
        while not res[x] and st[x] != s0:
            st[x] = s0; path.append(x); x = a[x]
        if not res[x]:
            i = path.index(x); L = len(path) - i
            for y in path[i:]:
                res[y] = L
            path = path[:i]
        for y in reversed(path):
            res[y] = res[a[y]] + 1
    return ' '.join(map(str, res[1:]))

def brute(s):
    v = ints(s); n = v[0]; a = [0] + v[1:]; out = []
    for i in range(1, n + 1):
        seen = set(); x = i
        while x not in seen:
            seen.add(x); x = a[x]
        out.append(len(seen))
    return ' '.join(map(str, out))

def sol(s):
    v = ints(s); n, m = v[0], v[1]; es = [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(m)]
    comp, c = scc(n, es); hasin = [False] * (c + 1)

            hasin[comp[b]] = True

def brute(s):
    v = ints(s); n, m = v[0], v[1]
    R = [[i == j for j in range(n + 1)] for i in range(n + 1)]
    for i in range(m):
        R[v[2 + 2 * i]][v[3 + 2 * i]] = True
    for x in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if R[i][x] and R[x][j]:
                    R[i][j] = True
    for t in range(1, n + 1):
        for comb in itertools.combinations(range(1, n + 1), t):

                return str(t)

--- code
def gen(r, k):
    n = sz(r, k, 1000, lo=1, tiny=7)
    es = tree(r, n, k) if k % 2 and n > 1 else []
    es = list(es)
    ex = r.randint(0, 3) if k % 2 else r.randint(0, int(1.5 * n))
    S = {(min(a, b), max(a, b)) for a, b in es}
    for _ in range(ex):
        if n < 3:
            break
        a, b = r.randint(1, n), r.randint(1, n)
        if a != b and (min(a, b), max(a, b)) not in S and len(S) < 3000:
            S.add((min(a, b), max(a, b))); es.append((a, b))
    if not k % 2:
        es = graph(r, n, r.randint(0, min(3000, int(1.3 * n))))
    return '%d %d\n' % (n, len(es)) + el(es)

def sol(s):
    v = ints(s); n, m = v[0], v[1]; g = adj(n, [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(m)]); best = INF
    for s0 in range(1, n + 1):
        d = {s0: 0}; par = {s0: 0}; q = deque([s0])
        while q:
            x = q.popleft()
            if 2 * d[x] + 1 >= best:
                break
            for y in g[x]:
                if y not in d:
                    d[y] = d[x] + 1; par[y] = x; q.append(y)
                elif y != par[x]:
                    best = min(best, d[x] + d[y] + 1)
    return str(-1 if best == INF else best)

def brute(s):
    v = ints(s); n, m = v[0], v[1]; g = adj(n, [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(m)]); best = [INF]
    def f(start, x, seen, L):
        for y in g[x]:
            if y == start and L >= 2:
                best[0] = min(best[0], L + 1)
            elif y > start and y not in seen:
                f(start, y, seen | {y}, L + 1)
    for st in range(1, n + 1):
        f(st, st, {st}, 0)
    return str(-1 if best[0] == INF else best[0])

def gen(r, k):
    n = sz(r, k, 100000, lo=2)
    if k % 3 == 0 and n > 3:
        c = r.randint(1, 3); es = [(c, i) for i in range(1, n + 1) if i != c]
        es = [e for e in es if r.random() < .9] + [(a, b) for a, b in graph(r, n, r.randint(0, n // 4)) if a != c and b != c]
        es = list({(min(a, b), max(a, b)) for a, b in es}); es = list(es)

def sol(s):
    v = ints(s); n, m = v[0], v[1]; deg = [0] * (n + 1); E = [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(m)]
    for a, b in E:
        deg[a] += 1; deg[b] += 1
    cls = defaultdict(int)
    for x in range(1, n + 1):
        cls[deg[x]] += 1
    ds = sorted(cls, reverse=True)[:3]; ed = defaultdict(int)
    for a, b in E:

            if pairs == 0:

            best = max(best, da + db - (1 if pairs <= e else 0))
