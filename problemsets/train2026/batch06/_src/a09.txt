def sol(s):
    t=list(map(int,s.split())); n=t[0]; a=[t[1+i*n:1+(i+1)*n] for i in range(n)]
    NEG=-1; dp=[[NEG]*n for _ in range(n)]; dp[0][0]=a[0][0]
    for st in range(1,2*n-1):
        nd=[[NEG]*n for _ in range(n)]
        for x1 in range(n):
            y1=st-x1
            if not 0<=y1<n: continue
            for x2 in range(n):
                y2=st-x2
                if not 0<=y2<n: continue
                b=NEG
                for p1 in (x1,x1-1):
                    for p2 in (x2,x2-1):
                        if 0<=p1<n and 0<=p2<n and dp[p1][p2]>b: b=dp[p1][p2]
                if b<0: continue
                v=a[x1][y1]+(a[x2][y2] if x1!=x2 else 0)
                nd[x1][x2]=b+v
        dp=nd
    return dp[n-1][n-1]

def brute(s):
    t=list(map(int,s.split())); n=t[0]; a=[t[1+i*n:1+(i+1)*n] for i in range(n)]
    paths=[]
    def go(i,j,cur):
        cur=cur+[(i,j)]
        if i==n-1 and j==n-1: paths.append(cur); return
        if i+1<n: go(i+1,j,cur)
        if j+1<n: go(i,j+1,cur)
    go(0,0,[])
    best=0
    for p in paths:
        for q in paths:
            cells=set(p)|set(q)
            best=max(best,sum(a[i][j] for i,j in cells))
    return best

Король стоит в левом верхнем углу доски (n+1)×(m+1) и должен попасть в правый нижний. За ход он идёт на клетку вправо, вниз или по диагонали вправо-вниз. Сколько существует маршрутов?

**Ввод.** Два числа n и m (1 ≤ n, m ≤ 1000).

**Вывод.** Число маршрутов по модулю 10^9 + 7.

def sol(s):
    n,m=map(int,s.split()); prev=[1]*(m+1)
    for i in range(1,n+1):
        cur=[1]+[0]*m
        for j in range(1,m+1): cur[j]=(prev[j]+cur[j-1]+prev[j-1])%M
        prev=cur
    return prev[m]
#B
def brute(s):
    n,m=map(int,s.split())
    def go(i,j):
        if i==n and j==m: return 1
        t=0
        if i<n: t+=go(i+1,j)
        if j<m: t+=go(i,j+1)
        if i<n and j<m: t+=go(i+1,j+1)
        return t
    return go(0,0)%M
#### Покраска домов
#S
Улицу из n домов красят в один из трёх цветов. Покраска дома i в цвет j стоит c_ij. Соседние дома должны быть разных цветов. Какова наименьшая суммарная стоимость?

**Ввод.** В первой строке n (1 ≤ n ≤ 10^5), далее n строк по три числа c_i1, c_i2, c_i3 (1 ≤ c_ij ≤ 1000).

**Вывод.** Наименьшая стоимость.

#A
def sol(s):
    t=list(map(int,s.split())); n=t[0]; a,b,c=t[1:4]
    for i in range(1,n):
        x,y,z=t[1+3*i:4+3*i]
        a,b,c=x+min(b,c),y+min(a,c),z+min(a,b)
    return min(a,b,c)
#B
def brute(s):
    from itertools import product
    t=list(map(int,s.split())); n=t[0]; best=10**18
    for col in product(range(3),repeat=n):
        if any(col[i]==col[i+1] for i in range(n-1)): continue
        best=min(best,sum(t[1+3*i+col[i]] for i in range(n)))
    return best
