def sol(s):
    t=list(map(int,s.split())); n=t[0]; d=[t[1+i*n:1+(i+1)*n] for i in range(n)]
    INF=10**18; F=1<<n; dp=[[INF]*n for _ in range(F)]; dp[1][0]=0
    for m in range(1,F,2):
        row=dp[m]
        for v in range(n):
            c=row[v]
            if c>=INF: continue
            for u in range(1,n):
                if not m>>u&1:
                    nm=m|1<<u
                    if c+d[v][u]<dp[nm][u]: dp[nm][u]=c+d[v][u]
    return min(dp[F-1][v]+d[v][0] for v in range(1,n))

def brute(s):
    from itertools import permutations
    t=list(map(int,s.split())); n=t[0]; d=[t[1+i*n:1+(i+1)*n] for i in range(n)]
    best=10**18
    for p in permutations(range(1,n)):
        c=d[0][p[0]]+sum(d[p[i]][p[i+1]] for i in range(n-2))+d[p[-1]][0]
        best=min(best,c)
    return best

def sol(s):
    t=list(map(int,s.split())); n,k=t[0],t[1]; a=t[2:2+n]; tot=sum(a)
    if tot%k: return "NO"
    T=tot//k
    if max(a)>T: return "NO"
    F=1<<n; dp=[-1]*F; dp[0]=0
    for m in range(F):
        if dp[m]<0: continue
        for i in range(n):
            if not m>>i&1 and dp[m]+a[i]<=T:
                nm=m|1<<i
                if dp[nm]<0: dp[nm]=(dp[m]+a[i])%T
    return "YES" if dp[F-1]==0 else "NO"

def brute(s):
    t=list(map(int,s.split())); n,k=t[0],t[1]; a=t[2:2+n]; tot=sum(a)
    if tot%k: return "NO"
    T=tot//k; sums=[0]*k
    def go(i):
        if i==n: return all(x==T for x in sums)
        for g in range(k):
            if sums[g]+a[i]<=T:
                sums[g]+=a[i]
                if go(i+1): sums[g]-=a[i]; return True
                sums[g]-=a[i]
        return False
    return "YES" if go(0) else "NO"

def sol(s):
    t=list(map(int,s.split())); n,C=t[0],t[1]; w=t[2:2+n]; F=1<<n
    INF=(10**9,0); dp=[INF]*F; dp[0]=(1,0)

            if bins[b]+w[i]<=C:
                bins[b]+=w[i]; go(i+1); bins[b]-=w[i]
        bins.append(w[i]); go(i+1); bins.pop()
    go(0)
    return best[0]

def sol(s):
    import math
    t=list(map(int,s.split())); n=t[0]; a=t[1:1+n]
    sq=lambda x: math.isqrt(x)**2==x
    ok=[[sq(a[i]+a[j]) for j in range(n)] for i in range(n)]
    F=1<<n; dp=[[0]*n for _ in range(F)]
    for v in range(n): dp[1<<v][v]=1
    for m in range(1,F):
        for v in range(n):
            c=dp[m][v]
            if not c: continue
            for u in range(n):
                if not m>>u&1 and ok[v][u]: dp[m|1<<u][u]+=c
    return sum(dp[F-1])

def sol(s):
    n,m=map(int,s.split()); 
    dp={0:1}
    for c in range(m):
        for rr in range(n):
            nd={}

def brute(s):
    n,m=map(int,s.split()); g=[[0]*m for _ in range(n)]
    def go(p):
        while p<n*m and g[p//m][p%m]: p+=1
        if p==n*m: return 1
        r,c=p//m,p%m; t=0
        g[r][c]=1
        if c+1<m and not g[r][c+1]: g[r][c+1]=1; t+=go(p+1); g[r][c+1]=0
        if r+1<n and not g[r+1][c]: g[r+1][c]=1; t+=go(p+1); g[r+1][c]=0
        g[r][c]=0
        return t
    return go(0)%M
