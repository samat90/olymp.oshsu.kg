#### Торговля с паузой
#S
Известны цены зерна на n дней. Каждый день Каныкей может купить одну меру зерна, продать её или ничего не делать, но держать больше одной меры нельзя. После продажи на следующий день покупать нельзя (пауза один день). Какую наибольшую прибыль можно получить?

**Ввод.** В первой строке n (1 ≤ n ≤ 10^5), во второй n цен p_i (1 ≤ p_i ≤ 10^4).

**Вывод.** Наибольшая прибыль.

1 2 3 0 2

#A
def sol(s):
    t=list(map(int,s.split())); p=t[1:1+t[0]]
    hold=-10**18; free=0; cool=0
    for x in p:
        nh=max(hold,free-x); nf=max(free,cool); nc=hold+x
        hold,free,cool=nh,nf,nc
    return max(free,cool,0)
#B
def brute(s):
    t=list(map(int,s.split())); n=t[0]; p=t[1:1+n]
    def go(i,state,blocked):
        if i==n: return 0
        best=go(i+1,state,False)
        if state==0 and not blocked: best=max(best,-p[i]+go(i+1,1,False))
        if state==1: best=max(best,p[i]+go(i+1,0,True))
        return best
    return go(0,0,False)

def brute(s):

    def go(i,holding,left):
        if i==n: return 0
        best=go(i+1,holding,left)
        if holding is None:
            if left>0: best=max(best,go(i+1,p[i],left-1)-p[i])
        else: best=max(best,p[i]+go(i+1,None,left))
        return best
    return go(0,None,K)

**Ввод.** В первой строке n (1 ≤ n ≤ 10^5), во второй n чисел a_i (0 ≤ a_i ≤ 1000).

**Вывод.** Наименьшее число прыжков или −1.

def sol(s):
    t=list(map(int,s.split())); n=t[0]; a=t[1:1+n]
    i=0; end=0; far=0; jumps=0
    if n==1: return 0
    for i in range(n-1):
        far=max(far,i+a[i])
        if i==end:
            if far<=i: return -1
            jumps+=1; end=far
            if end>=n-1: return jumps
    return jumps if end>=n-1 else -1

def brute(s):
    from collections import deque
    t=list(map(int,s.split())); n=t[0]; a=t[1:1+n]
    d={0:0}; q=deque([0])
    while q:
        x=q.popleft()
        for y in range(x+1,min(n-1,x+a[x])+1):
            if y not in d: d[y]=d[x]+1; q.append(y)
    return d.get(n-1,-1)

def sol(s):
    d=list(map(int,s.split()[0])); n=len(d)
    from functools import lru_cache
    @lru_cache(None)
    def f(pos,last,tight,started):
        if pos==n: return 1 if started else 0
        res=0; lim=d[pos] if tight else 9
        for x in range(lim+1):
            if started and x==last: continue
            ns=started or x>0
            res+=f(pos+1,x if ns else -1,tight and x==lim,ns)
        return res
    return f(0,-1,True,False)

def sol(s):
    t=list(map(int,s.split())); n=t[0]; w=t[1:1+n]; g=[[] for _ in range(n)]
    for i in range(n-1):
        a,b=t[1+n+2*i]-1,t[2+n+2*i]-1; g[a].append(b); g[b].append(a)

#### Сторожа в дереве
#S
Селения образуют дерево из n вершин. Сторож, поставленный в селение, охраняет его и все соседние селения. Какое наименьшее число сторожей нужно, чтобы охранялись все селения?

**Ввод.** В первой строке n (1 ≤ n ≤ 10^5), далее n−1 строк с рёбрами a, b.

**Вывод.** Наименьшее число сторожей.

def sol(s):
    t=list(map(int,s.split())); n=t[0]; g=[[] for _ in range(n)]
    for i in range(n-1):
        a,b=t[1+2*i]-1,t[2+2*i]-1; g[a].append(b); g[b].append(a)
    par=[-1]*n; order=[0]; seen=[False]*n; seen[0]=True
    for v in order:
        for u in g[v]:
            if not seen[u]: seen[u]=True; par[u]=v; order.append(u)
    INF=10**9
    a=[1]*n   # v has guard
    b=[0]*n   # v covered by a child (no guard on v)
    c=[0]*n   # v not covered yet (children all covered)
    ch=[[] for _ in range(n)]
    for v in order[1:]: ch[par[v]].append(v)
    for v in reversed(order):
        if not ch[v]:
            a[v]=1; b[v]=INF; c[v]=0
            continue
        a[v]=1+sum(min(a[u],b[u],c[u]) for u in ch[v])
        c[v]=sum(b[u] for u in ch[v])
        base=sum(min(a[u],b[u]) for u in ch[v])
        extra=min(max(0,a[u]-min(a[u],b[u])) for u in ch[v])
        b[v]=base+extra
    return min(a[0],b[0])

def brute(s):
    t=list(map(int,s.split())); n=t[0]; E=[(t[1+2*i]-1,t[2+2*i]-1) for i in range(n-1)]
    nb=[1<<i for i in range(n)]
    for a,b in E: nb[a]|=1<<b; nb[b]|=1<<a
    best=n
    for m in range(1<<n):
        cov=0
        for i in range(n):
            if m>>i&1: cov|=nb[i]
        if cov==(1<<n)-1: best=min(best,bin(m).count('1'))
    return best
