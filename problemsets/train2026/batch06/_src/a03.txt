#### Самый длинный подъём
#S
Дан массив из n чисел. Найдите длину самой длинной строго возрастающей подпоследовательности (элементы берутся в исходном порядке, не обязательно подряд).

**Ввод.** В первой строке n (1 ≤ n ≤ 10^5), во второй n чисел a_i (1 ≤ a_i ≤ 10^9).

**Вывод.** Длина.

8
5 2 8 6 3 6 9 7

    hi=6 if k<4 else 10**9
    a=L(r,n,1,hi)

    return "%d\n%s\n"%(n,' '.join(map(str,a)))
#A
def sol(s):
    from bisect import bisect_left
    t=list(map(int,s.split())); a=t[1:1+t[0]]; d=[]
    for x in a:
        i=bisect_left(d,x)
        if i==len(d): d.append(x)
        else: d[i]=x
    return len(d)
#B
def brute(s):
    t=list(map(int,s.split())); a=t[1:1+t[0]]; n=len(a); best=0
    for m in range(1<<n):
        v=[a[i] for i in range(n) if m>>i&1]
        if all(v[i]<v[i+1] for i in range(len(v)-1)): best=max(best,len(v))
    return best

def sol(s):
    from bisect import bisect_left
    t=list(map(int,s.split())); a=t[1:1+t[0]]; n=len(a)
    def lis(a):
        d=[]; res=[]
        for x in a:
            i=bisect_left(d,x)
            if i==len(d): d.append(x)
            else: d[i]=x
            res.append(i+1)
        return res

    return best

def sol(s):
    a,b=s.split()[:2]; n,m=len(a),len(b)
    F=[[0]*(m+2) for _ in range(n+2)]
    for i in range(n-1,-1,-1):
        for j in range(m-1,-1,-1):
            F[i][j]=F[i+1][j+1]+1 if a[i]==b[j] else max(F[i+1][j],F[i][j+1])
    al=sorted(set(a)&set(b))
    nxa=[{} for _ in range(n+1)]; nxb=[{} for _ in range(m+1)]
    cur={}
    for i in range(n-1,-1,-1):
        cur=dict(cur); cur[a[i]]=i; nxa[i]=cur
    cur={}
    for j in range(m-1,-1,-1):
        cur=dict(cur); cur[b[j]]=j; nxb[j]=cur
    i=j=0; out=[]; need=F[0][0]
    while need>0:
        for c in al:
            if c in nxa[i] and c in nxb[j]:
                p=nxa[i][c]; q=nxb[j][c]
                if F[p+1][q+1]==need-1:
                    out.append(c); i=p+1; j=q+1; need-=1; break
    return "%d\n%s"%(len(out),''.join(out))

def sol(s):
    w=s.split()[0]; n=len(w); best=1
    pal=[[False]*n for _ in range(n)]
