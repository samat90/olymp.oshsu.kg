=== 93 | Мост между островами | Графы | 2
--- statement
На карте $h \times w$ символ `#` — суша, `.` — вода. Остров — группа клеток суши, соседних по стороне. На карте ровно два острова. Сколько водных клеток нужно превратить в сушу, чтобы острова стали одним?

**Ввод.** В первой строке $h$ и $w$ ($1 \le h\cdot w \le 10^5$). Далее $h$ строк по $w$ символов.

**Вывод.** Одно число.
--- sample
4 5
##...
#....
....#
...##
--- code
def gen(r, k):
    h, w = hw(r, k, 100000, tiny=30)
    if h * w < 3:
        h, w = 1, 3
    while True:
        cells = [(i, j) for i in range(h) for j in range(w)]
        a, b = r.choice(cells), r.choice(cells)
        if abs(a[0] - b[0]) + abs(a[1] - b[1]) < 2:
            continue
        A, B = {a}, {b}
        def near(c, S):
            return any((c[0] + x, c[1] + y) in S for x, y in D4)
        for S, T in ((A, B), (B, A)):
            for _ in range(r.randint(0, h * w // 8)):
                c = r.choice(list(S)) if len(S) < 50 else r.choice(list(S))
                x, y = r.choice(D4); n_ = (c[0] + x, c[1] + y)
                if 0 <= n_[0] < h and 0 <= n_[1] < w and n_ not in S and n_ not in T and not near(n_, T):
                    S.add(n_)
        g = [['.'] * w for _ in range(h)]
        for i, j in A | B:
            g[i][j] = '#'
        g = [''.join(x) for x in g]
        if len(set(grid_labels(g).values())) == 2:
            return '%d %d\n' % (h, w) + '\n'.join(g)

def sol(s):
    h, w, g = rg(s)
    start = next((i, j) for i in range(h) for j in range(w) if g[i][j] == '#')
    isl = {start}; st = [start]
    while st:
        x, y = st.pop()
        for a, b in D4:
            c = (x + a, y + b)
            if 0 <= c[0] < h and 0 <= c[1] < w and g[c[0]][c[1]] == '#' and c not in isl:
                isl.add(c); st.append(c)
    d = {c: 0 for c in isl}; q = deque(isl)
    while q:
        x, y = q.popleft()
        for a, b in D4:
            c = (x + a, y + b)
            if 0 <= c[0] < h and 0 <= c[1] < w and c not in d:
                if g[c[0]][c[1]] == '#':
                    return str(d[(x, y)])
                d[c] = d[(x, y)] + 1; q.append(c)

def brute(s):
    h, w, g = rg(s); lab = grid_labels(g); items = list(lab.items())
    return str(min(abs(p[0] - q[0]) + abs(p[1] - q[1]) - 1 for p, l1 in items for q, l2 in items if l1 != l2))
