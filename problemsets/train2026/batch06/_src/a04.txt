def sol(s):
    w=s.split()[0]; n=len(w)
    d=[[0]*n for _ in range(n)]
    for i in range(n-2,-1,-1):
        for j in range(i+1,n):
            d[i][j]=d[i+1][j-1] if w[i]==w[j] else min(d[i+1][j],d[i][j-1])+1

def brute(s):
    w=s.split()[0]
    from collections import deque
    seen={w}; q=deque([(w,0)])
    al=sorted(set(w))
    while q:
        x,d=q.popleft()
        if x==x[::-1]: return d
        for i in range(len(x)+1):
            for c in al:
                y=x[:i]+c+x[i:]
                if y not in seen: seen.add(y); q.append((y,d+1))

def sol(s):
    a,b,c=s.split()[:3]; n,m=len(a),len(b)
    if n+m!=len(c): return "NO"
    dp=[False]*(m+1); dp[0]=True

        for j in range(m+1):
            if i==0 and j==0: continue
            ok=False
            if i>0 and dp[j] and a[i-1]==c[i+j-1]: ok=True

def sol(s):

def gen(r,k):

#A
def sol(s):
    t=s.split(); n=int(t[0]); ws=sorted(t[1:1+n],key=len); dp={}; best=1
    for w in ws:
        b=1
        for i in range(len(w)):
            x=w[:i]+w[i+1:]
            if x in dp and dp[x]+1>b: b=dp[x]+1
        dp[w]=b; best=max(best,b)
    return best
#B
def brute(s):
    t=s.split(); n=int(t[0]); ws=t[1:1+n]
    def pred(x,y):
        return len(y)==len(x)+1 and any(y[:i]+y[i+1:]==x for i in range(len(y)))
    best=[1]
    def go(w,d):
        best[0]=max(best[0],d)
        for y in ws:
            if pred(w,y): go(y,d+1)
    for w in ws: go(w,1)
    return best[0]
