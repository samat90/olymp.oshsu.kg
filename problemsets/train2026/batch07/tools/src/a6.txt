def gen(r, k):
    n = sz(r, k, 100000)
    if k % 2 == 0:
        es = bip_edges(r, n, r.randint(n // 2, 2 * n))
        if k % 4 == 0 and es and n > 2:
            a, b = r.sample(range(1, n + 1), 2)
            if (min(a, b), max(a, b)) not in set((min(x, y), max(x, y)) for x, y in es):
                es.append((a, b))
    else:
        es = graph(r, n, r.randint(0, int(1.2 * n)))
    return '%d %d\n' % (n, len(es)) + el(es)

def sol(s):
    v = ints(s); n, m = v[0], v[1]
    g = adj(n, [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(m)]); col = [-1] * (n + 1)
    for st in range(1, n + 1):
        if col[st] < 0:
            col[st] = 0; q = [st]
            while q:
                x = q.pop()
                for y in g[x]:
                    if col[y] < 0:
                        col[y] = col[x] ^ 1; q.append(y)
                    elif col[y] == col[x]:
                        return 'NO'
    return 'YES'

def brute(s):
    v = ints(s); n, m = v[0], v[1]; es = [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(m)]
    for mask in range(1 << n):
        if all((mask >> (a - 1) & 1) != (mask >> (b - 1) & 1) for a, b in es):
            return 'YES'
    return 'NO'

    m = r.randint(0, min(a * b, 5000, 3 * max(a, b)))

        es.add((r.randint(1, a), r.randint(1, b)))

def sol(s):
    v = ints(s); a, b, m = v[:3]; g = [[] for _ in range(a + 1)]
    for i in range(m):
        g[v[3 + 2 * i]].append(v[4 + 2 * i])
    mt = [0] * (b + 1); res = 0
    def try_(x, seen):
        stack = [(x, iter(g[x]))]; path = []
        while stack:
            u, it = stack[-1]
            adv = False
            for y in it:
                if not seen[y]:
                    seen[y] = True
                    path.append((u, y))
                    if mt[y] == 0:
                        for uu, yy in path:
                            mt[yy] = uu
                        return True
                    stack.append((mt[y], iter(g[mt[y]]))); adv = True
                    break
            if not adv:
                stack.pop()
                if path:
                    path.pop()
        return False
    for x in range(1, a + 1):
        if try_(x, [False] * (b + 1)):
            res += 1
    return str(res)

    def f(i, used):
        if i > a:
            return 0
        best = f(i + 1, used)
        for y in g[i]:
            if not used >> y & 1:
                best = max(best, 1 + f(i + 1, used | 1 << y))
