    for m in range(1<<n):
        if sum(a[i] for i in range(n) if m>>i&1)==S0:
            c=bin(m).count('1')
            if best<0 or c<best: best=c
    return best

def sol(s):
    t=list(map(int,s.split())); n=t[0]; p=[0]+t[1:1+n]; dp=[0]*(n+1)
    for x in range(1,n+1): dp[x]=max(dp[x-j]+p[j] for j in range(1,x+1))
    return dp[n]

def brute(s):
    t=list(map(int,s.split())); n=t[0]; p=[0]+t[1:1+n]
    def go(rem):
        if rem==0: return 0
        return max(p[j]+go(rem-j) for j in range(1,rem+1))
    return go(n)
