**Ввод.** В первой строке n (2 ≤ n ≤ 10^5), во второй n высот (1 ≤ h_i ≤ 10^4).

**Вывод.** Минимальная затраченная энергия.

#A
def sol(s):
    t=list(map(int,s.split())); n=t[0]; h=t[1:]
    dp=[0]*n
    for i in range(1,n):
        dp[i]=dp[i-1]+abs(h[i]-h[i-1])
        if i>=2: dp[i]=min(dp[i],dp[i-2]+abs(h[i]-h[i-2]))
    return dp[-1]
#B
def brute(s):
    t=list(map(int,s.split())); n=t[0]; h=t[1:]
    best=[10**18]
    def go(i,c):
        if i==n-1: best[0]=min(best[0],c); return
        for d in (1,2):
            if i+d<n: go(i+d,c+abs(h[i]-h[i+d]))
    go(0,0); return best[0]

#A
def sol(s):
    t=list(map(int,s.split())); m,X=t[0],t[1]; c=t[2:2+m]
    INF=10**9; dp=[INF]*(X+1); dp[0]=0
    for x in range(1,X+1):
        b=INF
        for v in c:
            if v<=x and dp[x-v]+1<b: b=dp[x-v]+1
        dp[x]=b
    return dp[X] if dp[X]<INF else -1
#B
def brute(s):
    t=list(map(int,s.split())); m,X=t[0],t[1]; c=t[2:2+m]
    best=[10**9]
    def go(i,rem,cnt):
        if rem==0: best[0]=min(best[0],cnt); return
        if i==m: return
        for q in range(rem//c[i]+1): go(i+1,rem-q*c[i],cnt+q)
    go(0,X,0)
    return best[0] if best[0]<10**9 else -1
