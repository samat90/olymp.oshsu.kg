def gen(r, k):

def gen(r, k):
    n = sz(r, k, 100000); q = r.randint(1, 2 * n)
    cls = r.randint(1, max(1, n // 3)); lab = [r.randint(1, cls) for _ in range(n + 1)]
    out = []
    for _ in range(q):
        a, b = r.randint(1, n), r.randint(1, n)
        for _ in range(5):
            if r.random() < .5:
                b = r.randint(1, n)
            if (lab[a] == lab[b]) == (r.random() < .5):
                break
        out.append('%d %s %d' % (a, '=' if lab[a] == lab[b] else '!=', b))
    if n >= 3 and (k % 3 == 0 if k else r.random() < .5):
        x, y, z = r.sample(range(1, n + 1), 3)
        for c in ('%d = %d' % (x, y), '%d = %d' % (y, z), '%d != %d' % (x, z)):
            out.insert(r.randrange(len(out) + 1), c)
    return '%d %d\n' % (n, len(out)) + '\n'.join(out)

def sol(s):
    L = s.split('\n'); n, q = map(int, L[0].split()); d = DSU(n); ne = []
    for i in range(q):
        a, op, b = L[1 + i].split(); a = int(a); b = int(b)
        if op == '=':
            d.u(a, b)
        else:
            ne.append((a, b))
    return 'NO' if any(d.f(a) == d.f(b) for a, b in ne) else 'YES'

    lab = comps_bf(n, eq)

=== 34 | Убрать камни | Графы | 1
--- statement
На клетчатой плоскости лежат $n$ камней, каждый в своей клетке. Камень можно убрать, если в той же строке или в том же столбце лежит другой ещё не убранный камень. Какое наибольшее число камней можно убрать?

**Ввод.** В первой строке $n$ ($1 \le n \le 10^5$). Далее $n$ строк с парами $x\ y$ ($1 \le x, y \le 10^9$) — различными клетками.

**Вывод.** Одно число.
--- sample
5
1 1
1 3
3 1
3 3
5 7
--- code
def gen(r, k):
    n = sz(r, k, 100000, lo=1)
    C = max(2, int(n ** .5 * r.choice([1, 1.5, 3]))); C = min(C, 10**9)
    n = min(n, C * C); P = set()
    while len(P) < n:
        P.add((r.randint(1, C), r.randint(1, C)))
    P = list(P); r.shuffle(P)
    sc = r.choice([1, 1000, 10**7]) if k else 1
    return '%d\n' % n + el([(x * sc, y * sc) for x, y in P])

def sol(s):
    v = ints(s); n = v[0]; d = DSU(2 * n); R = {}; C = {}; c = n
    for i in range(n):
        x, y = v[1 + 2 * i], v[2 + 2 * i]
        for mp, key in ((R, x), (C, y)):
            if key in mp:
                if d.u(mp[key], i + 1):
                    c -= 1
            else:
                mp[key] = i + 1
    return str(n - c)

def brute(s):
    v = ints(s); n = v[0]; P = [(v[1 + 2 * i], v[2 + 2 * i]) for i in range(n)]
    @__import__('functools').lru_cache(None)
    def f(mask):
        best = 0
        for i in range(n):
            if mask >> i & 1 and any(j != i and mask >> j & 1 and (P[j][0] == P[i][0] or P[j][1] == P[i][1]) for j in range(n)):
                best = max(best, 1 + f(mask & ~(1 << i)))
        return best
    return str(f((1 << n) - 1))

def gen(r, k):

def bfs(g, s, n):
    d = [-1] * (n + 1); d[s] = 0; q = deque([s])

        x = q.popleft()

def sol(s):
    v = ints(s); n = v[0]
    g = adj(n, [(v[1 + 2 * i], v[2 + 2 * i]) for i in range(n - 1)])
    deg = [len(x) for x in g]; left = n; leaves = [i for i in range(1, n + 1) if deg[i] <= 1]

        left -= len(leaves); nl = []

def gen(r, k):
    n = sz(r, k, 50000, lo=1); q = sz(r, k, 50000, lo=1)
    return '%d %d\n' % (n, q) + el(tree(r, n, k)) + '\n' + el([(r.randint(1, n), r.randint(1, n)) for _ in range(q)])

def sol(s):
    v = ints(s); n, q = v[0], v[1]
    g = adj(n, [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(n - 1)])
    LG = max(1, n.bit_length()); up = [[0] * (n + 1) for _ in range(LG)]; dep = [0] * (n + 1)
    st = [1]; seen = [False] * (n + 1); seen[1] = True; up[0][1] = 1
    while st:
        x = st.pop()
        for y in g[x]:
            if not seen[y]:
                seen[y] = True; up[0][y] = x; dep[y] = dep[x] + 1; st.append(y)
    for j in range(1, LG):
        a, b = up[j - 1], up[j]
        for x in range(1, n + 1):
            b[x] = a[a[x]]
    out = []; p = 2 + 2 * (n - 1)
    for i in range(q):
        a, b = v[p + 2 * i], v[p + 2 * i + 1]; x, y = a, b
        if dep[x] < dep[y]:
            x, y = y, x
        df = dep[x] - dep[y]
        for j in range(LG):
            if df >> j & 1:
                x = up[j][x]
        if x != y:
            for j in range(LG - 1, -1, -1):
                if up[j][x] != up[j][y]:
                    x, y = up[j][x], up[j][y]
            x = up[0][x]
        out.append(dep[a] + dep[b] - 2 * dep[x])
    return '\n'.join(map(str, out))

    g = adj(n, [(v[2 + 2 * i], v[3 + 2 * i]) for i in range(n - 1)]); p = 2 + 2 * (n - 1); out = []
    for i in range(q):
        a, b = v[p + 2 * i], v[p + 2 * i + 1]
        d = {a: 0}; qu = deque([a])
        while qu:
            x = qu.popleft()
            for y in g[x]:

                    d[y] = d[x] + 1; qu.append(y)

def gen(r, k):
    n = sz(r, k, 100000, lo=1)
    e = tree(r, n, k); h = r.randint(1, n); g = adj(n, e); par = [0] * (n + 1); par[h] = -1; st = [h]
    while st:
        x = st.pop()
        for y in g[x]:
            if par[y] == 0:
                par[y] = x; st.append(y)
    return '%d %d\n' % (n, h) + ' '.join(map(str, par[1:])) + '\n' + ' '.join(str(r.randint(0, r.choice([3, 1000]))) for _ in range(n))

def sol(s):
    v = ints(s); n, h = v[:2]; par = v[2:2 + n]; t = v[2 + n:2 + 2 * n]
    ch = [[] for _ in range(n + 1)]
    for i in range(n):
        if par[i] > 0:
            ch[par[i]].append(i + 1)
    best = 0; st = [(h, 0)]
    while st:
        x, d = st.pop(); best = max(best, d)
        for y in ch[x]:
            st.append((y, d + t[x - 1]))
    return str(best)

def brute(s):
    v = ints(s); n, h = v[:2]; par = v[2:2 + n]; t = v[2 + n:2 + 2 * n]; best = 0
    for i in range(1, n + 1):
        x = i; tot = 0
        while par[x - 1] > 0:
            x = par[x - 1]; tot += t[x - 1]
        best = max(best, tot)
    return str(best)
